# =========================================================
# تنظيف النص للمقارنة
# =========================================================

def normalize_for_check(text):

    text = text.lower()

    # إزالة المسافات الزائدة
    text = re.sub(r"\s+", " ", text)

    return text.strip()


# =========================================================
# تحديد هل النص غير مهم للأدوية
# =========================================================

def is_irrelevant_text(text):

    normalized = normalize_for_check(text)

    for word in IGNORE_WORDS:

        word_normalized = normalize_for_check(word)

        if word_normalized in normalized:
            return True

    return False


# =========================================================
# تحديد هل النص مرشح ليكون دواء
# =========================================================

def is_medicine_candidate(text):

    normalized = normalize_for_check(text)

    # -----------------------------------------
    # النصوص القصيرة جداً
    # -----------------------------------------

    if len(normalized) < 3:
        return False


    # -----------------------------------------
    # إذا كان نصاً معروفاً أنه غير مهم
    # -----------------------------------------

    if is_irrelevant_text(normalized):
        return False


    # -----------------------------------------
    # إذا كان النص أرقام فقط
    # -----------------------------------------

    if normalized.isdigit():
        return False


    # -----------------------------------------
    # إذا كان النص يحتوي أرقام فقط مع رموز
    # مثل:
    # 130130
    # 37
    # 29
    # -----------------------------------------

    if re.fullmatch(r"[\d\s:/\-.]+", normalized):
        return False


    # -----------------------------------------
    # إذا كان النص عنواناً أو رقم هاتف
    # -----------------------------------------

    if re.search(r"\d{7,}", normalized):
        return False


    # -----------------------------------------
    # كلمات غالباً تشير إلى تعليمات الجرعة
    # -----------------------------------------

    instruction_words = [

        "ساعة",
        "ساعات",

        "مرة",
        "مرات",

        "يوم",
        "أيام",

        "بعد الأكل",

        "قبل الأكل",

        "صباحا",
        "صباحًا",

        "مساء",

        "ليلاً",
        "ليلا",

    ]

    # إذا كان النص تعليمات فقط
    if any(word in normalized for word in instruction_words):

        # لكن لا نستبعده إذا كان يحتوي على اسم دواء
        # أو كلمة دوائية واضحة مع جرعة

        if not re.search(
            r"[a-zA-Z]{3,}",
            normalized
        ):
            return False


    # -----------------------------------------
    # النص الذي يحتوي أحرف إنجليزية
    # -----------------------------------------

    if re.search(r"[a-zA-Z]{3,}", normalized):
        return True


    # -----------------------------------------
    # النص العربي
    # -----------------------------------------

    # في هذه المرحلة نسمح للنص العربي أن يكون مرشحاً
    # لأن بعض أسماء الأدوية تكتب بالعربي

    if re.search(r"[\u0600-\u06FF]{3,}", normalized):
        return True


    return False


# =========================================================
# تشغيل OCR
# =========================================================

result = ocr.predict(str(image_path))


# =========================================================
# القوائم
# =========================================================

texts = []

medicine_candidates = []

other_texts = []


# =========================================================
# استخراج النصوص
# =========================================================

for res in result:

    rec_texts = res.get("rec_texts", [])
    rec_scores = res.get("rec_scores", [])


    for text, score in zip(rec_texts, rec_scores):

        # -----------------------------------------
        # فحص OCR
        # -----------------------------------------

        if not is_valid_text(text, score):
            continue


        # -----------------------------------------
        # تنظيف النص
        # -----------------------------------------

        cleaned = clean_text(text)

        if not cleaned:
            continue


        item = {
            "text": cleaned,
            "score": round(float(score), 4)
        }


        # -----------------------------------------
        # حفظ كل النصوص
        # -----------------------------------------

        texts.append(item)


        # -----------------------------------------
        # تصنيف النص
        # -----------------------------------------

        if is_medicine_candidate(cleaned):

            medicine_candidates.append(item)

        else:

            other_texts.append(item)


# =========================================================
# عرض كل النصوص
# =========================================================

print("\n" + "=" * 60)
print("كل النصوص التي قرأها OCR")
print("=" * 60)

    print(

dataset_item = {
