const PAGE_SIZE = 8;

// --- Helper Functions ---
const getMinPrice = (m: Medicine) => {
  const inStock = m.pharmacies.filter((p) => p.stock);
  const pool = inStock.length ? inStock : m.pharmacies;
  return Math.min(...pool.map((p) => p.price));
};

const checkIsAvailable = (m: Medicine) => m.pharmacies.some((p) => p.stock);

interface CatalogProps {
  searchQuery: string;
  onSearchChange: (query: string) => void;
}

const Catalog: React.FC<CatalogProps> = ({ searchQuery,onSearchChange }) => {
  // --- States ---
//   const [searchQuery, setSearchQuery] = useState('');
  const [selectedCat, setSelectedCat] = useState('الكل');
  const [selectedPrice, setSelectedPrice] = useState('الكل');
  const [inStockOnly, setInStockOnly] = useState(false);
  const [sortOption, setSortOption] = useState('popular');
  const [visibleCount, setVisibleCount] = useState(PAGE_SIZE);
  const [openPharmacies, setOpenPharmacies] = useState<Record<number, boolean>>({});
  const [isFiltersOpen, setIsFiltersOpen] = useState(false);

  // --- Filtering & Sorting ---
  const filteredMedicines = useMemo(() => {
    const list = MEDICINES.filter((m) => {
      const matchesCat = selectedCat === 'الكل' || m.cat === selectedCat;
      const matchesQuery = !searchQuery || m.name.includes(searchQuery);
      const matchesStock = !inStockOnly || checkIsAvailable(m);
      let matchesPrice = true;

      if (selectedPrice !== 'الكل') {
        const [min, max] = selectedPrice.split('-').map(Number);
        const p = getMinPrice(m);
        matchesPrice = p >= min && p <= max;
      }
      return matchesCat && matchesQuery && matchesStock && matchesPrice;
    });

    if (sortOption === 'price-asc') list.sort((a, b) => getMinPrice(a) - getMinPrice(b));
    else if (sortOption === 'price-desc') list.sort((a, b) => getMinPrice(b) - getMinPrice(a));
    else if (sortOption === 'name') list.sort((a, b) => a.name.localeCompare(b.name, 'ar'));
    else list.sort((a, b) => b.popularity - a.popularity);

    return list;
  }, [searchQuery, selectedCat, selectedPrice, inStockOnly, sortOption]);

  const visibleMedicines = filteredMedicines.slice(0, visibleCount);

  // --- Handlers ---
  const handleCategoryChange = (cat: string) => {
    setSelectedCat(cat);
    setVisibleCount(PAGE_SIZE);
  };

  const handleReset = () => {
    onSearchChange('');
    setSelectedCat('الكل');
    setSelectedPrice('الكل');
    setInStockOnly(false);
    setSortOption('popular');
    setVisibleCount(PAGE_SIZE);
  };

  const togglePharmList = (idx: number) => {
    setOpenPharmacies((prev) => ({ ...prev, [idx]: !prev[idx] }));
  };

  return (
    <section className="section">
      
      <div className="container catalog-layout">
        
        {/* Filters sidebar */}
        <aside className={`filters  ${isFiltersOpen ? 'is-open' : ''}`} id="filtersPanel">
          <div className="filters__head">
            <h3>الفلاتر</h3>
            <button className="filters__reset" onClick={handleReset}>
              إعادة تعيين
            </button>
          </div>

          {/* Category Filter */}
          <div className="filter-group">
            <span className="filter-group__title">التصنيف</span>
            <div className="filter-options">
              {['الكل', 'مسكنات', 'مضادات حيوية', 'فيتامينات', 'أطفال', 'أمراض مزمنة', 'بشرة'].map((cat) => (
                <label key={cat} className="filter-check">
                  <input
                    type="radio"
                    name="cat"
                    value={cat}
                    checked={selectedCat === cat}
                    onChange={(e) => handleCategoryChange(e.target.value)}
                  />
                  <span>{cat === 'فيتامينات' ? 'فيتامينات ومكملات' : cat === 'أطفال' ? 'أدوية الأطفال' : cat === 'أمراض مزمنة' ? 'الأمراض المزمنة' : cat === 'بشرة' ? 'العناية بالبشرة' : cat}</span>
                </label>
              ))}
            </div>
          </div>

          {/* Price Filter */}
          <div className="filter-group">
            <span className="filter-group__title">السعر</span>
            <div className="filter-options">
              {[
                { label: 'كل الأسعار', value: 'الكل' },
                { label: 'أقل من 10 د.ل', value: '0-10' },
                { label: '10 – 25 د.ل', value: '10-25' },
                { label: 'أكثر من 25 د.ل', value: '25-1000' },
              ].map((p) => (
                <label key={p.value} className="filter-check">
                  <input
                    type="radio"
                    name="price"
                    value={p.value}
                    checked={selectedPrice === p.value}
                    onChange={(e) => {
                      setSelectedPrice(e.target.value);
                      setVisibleCount(PAGE_SIZE);
                    }}
                  />
                  <span>{p.label}</span>
                </label>
              ))}
            </div>
          </div>

          {/* Availability Filter */}
          <div className="filter-group">
            <span className="filter-group__title">التوفر</span>
            <label className="filter-check">
              <input
                type="checkbox"
                checked={inStockOnly}
                onChange={(e) => {
                  setInStockOnly(e.target.checked);
                  setVisibleCount(PAGE_SIZE);
                }}
              />
              <span>عرض المتوفر فقط</span>
            </label>
          </div>
        </aside>

        {/* Results */}
        <div className="catalog-results">
          <div className="results-toolbar ">
            {/* Category Chips */}
            <div className="chip-row">
              {['الكل', 'مسكنات', 'مضادات حيوية', 'فيتامينات', 'أطفال', 'أمراض مزمنة', 'بشرة'].map((cat) => (
                <button
                  key={cat}
                  className={`chip ${selectedCat === cat ? 'is-active' : ''}`}
                  onClick={() => handleCategoryChange(cat)}
                >
                  {cat}
                </button>
              ))}
            </div>

            {/* Sort Dropdown & Mobile Toggle */}
            <div className="sort-row">
              <button
                className="filters-toggle"
                onClick={() => setIsFiltersOpen((prev) => !prev)}
              >
                <svg viewBox="0 0 24 24" width="16" height="16">
                  <path fill="currentColor" d="M4 6h16v2H4V6Zm3 5h10v2H7v-2Zm4 5h2v2h-2v-2Z" />
                </svg>
                الفلاتر
              </button>
              <label htmlFor="sortSelect">ترتيب حسب</label>
              <select
                id="sortSelect"
                value={sortOption}
                onChange={(e) => setSortOption(e.target.value)}
              >
                <option value="popular">الأكثر طلبًا</option>
                <option value="price-asc">السعر: الأقل أولًا</option>
                <option value="price-desc">السعر: الأعلى أولًا</option>
                <option value="name">الاسم (أ - ي)</option>
              </select>
            </div>
          </div>

          {/* Grid View */}
          <div className="med-grid">
            {visibleMedicines.map((m, idx) => {
              const available = checkIsAvailable(m);
              const isOpen = !!openPharmacies[idx];
              const sortedPharmacies = [...m.pharmacies].sort((a, b) => a.price - b.price);

              return (
                <article className="med-card" key={idx}>
                  <div className="med-card__top">
                    <span className="med-card__icon">
                      <svg viewBox="0 0 24 24" width="24" height="24">
                        <path fill="currentColor" d="M4.9 19.1a5.2 5.2 0 0 1 0-7.35l6.85-6.85a5.2 5.2 0 1 1 7.35 7.35l-6.85 6.85a5.2 5.2 0 0 1-7.35 0Zm3.68-9.6-3.03 3.03a3.2 3.2 0 0 0 4.52 4.52l3.03-3.03-4.52-4.52Z"/>
                      </svg>
                    </span>
                    <span className={`med-card__stock ${available ? '' : 'out'}`}>
                      {available ? 'متوفر' : 'غير متوفر'}
                    </span>
                  </div>
                  <span className="med-card__cat">{m.cat}</span>
                  <div>
                    <div className="med-card__name">{m.name}</div>
                    <div className="med-card__meta">{m.meta}</div>
                  </div>
                  <div className="med-card__foot">
                    <span className="med-card__price">يبدأ من {getMinPrice(m).toFixed(2)} د.ل</span>
                  </div>
                  <button
                    type="button"
                    className={`med-card__pharm-toggle ${isOpen ? 'is-open' : ''}`}
                    onClick={() => togglePharmList(idx)}
                  >
                    <span>عرض الصيدليات ({m.pharmacies.length})</span>
                    <svg viewBox="0 0 24 24" width="14" height="14">
                      <path fill="currentColor" d="M12 15.5 5 8.5l1.4-1.4L12 12.7l5.6-5.6L19 8.5z"/>
                    </svg>
                  </button>
                  <ul className={`med-card__pharm-list ${isOpen ? 'is-open' : ''}`}>
                    {sortedPharmacies.map((p, pIdx) => (
                      <li key={pIdx} className={p.stock ? '' : 'out'}>
                        <span>
                          <span className="pharm-name">{p.name}</span>
                          <span className="pharm-city">{p.city}</span>
                        </span>
                        <span className="pharm-price">
                          {p.stock ? `${p.price.toFixed(2)} د.ل` : 'غير متوفر'}
                        </span>
                      </li>
                    ))}
                  </ul>
                </article>
              );
            })}
          </div>

          {/* Empty State */}
          {filteredMedicines.length === 0 && (
            <p className="empty-state">لا توجد أدوية مطابقة لبحثك أو الفلاتر المحددة.</p>
          )}

          {/* Load More Button */}
          {visibleCount < filteredMedicines.length && (
            <div className="load-more-wrap">
              <button
                className="btn btn--outline"
                onClick={() => setVisibleCount((prev) => prev + PAGE_SIZE)}
              >
                عرض المزيد من الأدوية
              </button>
            </div>
          )}
        </div>
      </div>
    </section>
  );
};

export default Catalog;
